cmake_minimum_required(VERSION 3.16)
project(delayed_gradient_cpu_mlp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_library(model src/dense_layer.cpp src/mnist.cpp)
target_include_directories(model PUBLIC src)
add_executable(sequential_baseline src/sequential_baseline.cpp)
target_link_libraries(sequential_baseline PRIVATE model)
add_executable(mnist_reader src/main.cpp src/worker.cpp)
target_link_libraries(mnist_reader PRIVATE model Threads::Threads)
add_executable(test_core tests/test_core.cpp)
target_link_libraries(test_core PRIVATE model Threads::Threads)
foreach(target model sequential_baseline mnist_reader test_core)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
  endif()
endforeach()
enable_testing()
add_test(NAME model_and_queue COMMAND test_core)
