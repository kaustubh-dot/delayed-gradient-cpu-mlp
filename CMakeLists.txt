cmake_minimum_required(VERSION 3.16)
project(delayed_gradient_cpu_mlp LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Threads REQUIRED)
add_executable(mlp src/main.cpp)
target_link_libraries(mlp PRIVATE Threads::Threads)
if(MSVC)
  target_compile_options(mlp PRIVATE /W4)
else()
  target_compile_options(mlp PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME mathematical_and_queue_contracts COMMAND mlp --self-test)
